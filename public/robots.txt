# sandstr — simulations of real clients live under /c/. They are deliberately
# kept out of indexes: a pixel-faithful /c/damus ranking for "Damus" would be
# the textbook confusion pattern, and the gallery is the page that should rank.
User-agent: *
Disallow: /c/

# Card renderers are NOT indexers, and the rule above was silently costing us
# every /c/ link preview on X: Twitterbot implements the robots.txt spec and
# renders no card at all when the URL is disallowed, which is why X's own
# troubleshooting page tells publishers to add a Twitterbot exception.
# facebookexternalhit behaves the same way. Neither feeds a search index, so
# letting them through changes nothing about what ranks for "Damus" — it only
# lets a shared /c/<client> show that client's card (public/og/<id>.png,
# generated by scripts/og-client-cards.mjs) instead of nothing.
#
# Note the group semantics: a named user-agent group REPLACES the `*` group for
# that crawler, so these two lines are the whole policy each of them sees.
# Discord, Telegram and Slack fetch previews without consulting this file at
# all, so they are deliberately not listed — a line for them would be decoration.
User-agent: Twitterbot
Allow: /

User-agent: facebookexternalhit
Allow: /

# With no inbound links yet, this is the only active signal that the site
# exists. It lists "/" alone, for the reason above.
Sitemap: https://sandstr.app/sitemap.xml
